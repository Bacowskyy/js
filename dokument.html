import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.TreeSet;
import java.util.List;
import java.util.Set;

public class ListaObecnosci {
    public static void main(String[] args) {

        List<String> odbicia = new ArrayList<>();

        odbicia.add("Ola");
        odbicia.add("Jan");
        odbicia.add("Ola");
        odbicia.add("Piotr");
        odbicia.add("Jan");
        odbicia.add("Ewa");
        odbicia.add("Marta");

        System.out.println("Odbić karty: " + odbicia.size());

        Set<String> obecni = new LinkedHashSet<>();

        for (String imie : odbicia) {
            if (!obecni.add(imie)) {
                System.out.println("Ponowne odbicie: " + imie);
            }
        }

        Set<String> alfabetycznie = new TreeSet<>(obecni);

        System.out.println("Obecni (kolejność przyjścia): " + obecni);
        System.out.println("Obecni (alfabetycznie): " + alfabetycznie);
        System.out.println("Liczba obecnych: " + obecni.size());
        System.out.println("Czy Marek jest obecny? " + obecni.contains("Marek"));
    }
}
